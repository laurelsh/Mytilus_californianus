# Command log: WGS read processing, SNP calling, and diversity / Ne (Table S1)
#
# All steps were run in a terminal or as SLURM jobs on the lab server. Paths are placeholders (<...>).
# Samples (WGS, Psomagen, 150-bp paired-end): mc9a mc11a mc14a mc15a mc18a mc19a mc24a (= MC09 ... MC24)
# <WD> is the WGS working directory; <REF_FA> is mytilus_californianus_genome.fasta (GCF_021869535.1).

SAMPLES=(mc9a mc11a mc14a mc15a mc18a mc19a mc24a)


######################################################################
## 1. Adapter and quality trimming (Trim Galore v0.6.10, Cutadapt v2.6)
######################################################################
# Defaults, as recorded in the trimming reports: Phred quality cutoff 20; adapter auto-detected
# (Illumina TruSeq, AGATCGGAAGAGC); stringency 1 bp; error rate 0.1; pairs removed if either read < 20 bp.

conda activate trim-galore
trim_command=""
for s in "${SAMPLES[@]}"; do
  trim_command="$trim_command --paired <RAW_DIR>/${s}/${s}_1.fastq.gz <RAW_DIR>/${s}/${s}_2.fastq.gz"
done
trim_galore --output_dir <WD>/trimmed $trim_command
# Output: <WD>/trimmed/${s}_1_val_1.fq.gz, ${s}_2_val_2.fq.gz


######################################################################
## 2. Bowtie2 mapping, fixmate, duplicate removal, index (SLURM array, one task per sample)
######################################################################
# <SAMPLES_FILE>: one sample ID per line (mc9a, mc11a, ...)
# Bowtie2 v2.4.1 and Samtools v1.13. Index built once with: bowtie2-build <REF_FA> <BT2_PREFIX>

#!/usr/bin/env bash
#SBATCH --job-name=wgs_map_bt2
#SBATCH --cpus-per-task=24
#SBATCH --mem=64G
#SBATCH --time=1-00:00:00
#SBATCH --array=1-7
set -euo pipefail

SAMPLE="$(sed -n "${SLURM_ARRAY_TASK_ID}p" <SAMPLES_FILE>)"
R1="<WD>/trimmed/${SAMPLE}_1_val_1.fq.gz"
R2="<WD>/trimmed/${SAMPLE}_2_val_2.fq.gz"
MAPDIR="<WD>/mapping_bt2"
T=${SLURM_CPUS_PER_TASK}
mkdir -p "$MAPDIR/tmp_${SAMPLE}"

bowtie2 -x <BT2_PREFIX> -1 "$R1" -2 "$R2" -p "$T" 2> "$MAPDIR/${SAMPLE}.bt2.log" \
| samtools view -b -@ "$T" \
| samtools sort -n -@ "$T" -m 2G -T "$MAPDIR/tmp_${SAMPLE}/nsort" -o "$MAPDIR/${SAMPLE}.namesorted.bam"

samtools fixmate -@ "$T" -m "$MAPDIR/${SAMPLE}.namesorted.bam" "$MAPDIR/${SAMPLE}.fixmate.bam"
samtools sort    -@ "$T" -o "$MAPDIR/${SAMPLE}.coordsorted.bam" "$MAPDIR/${SAMPLE}.fixmate.bam"
samtools markdup -@ "$T" -r "$MAPDIR/${SAMPLE}.coordsorted.bam" "$MAPDIR/${SAMPLE}.mkdup.bam"   # -r: remove duplicates
samtools index   -@ "$T" "$MAPDIR/${SAMPLE}.mkdup.bam"
samtools flagstat -@ "$T" "$MAPDIR/${SAMPLE}.mkdup.bam" > "$MAPDIR/${SAMPLE}.flagstat.txt"

rm -f "$MAPDIR/${SAMPLE}.namesorted.bam" "$MAPDIR/${SAMPLE}.fixmate.bam" "$MAPDIR/${SAMPLE}.coordsorted.bam"


######################################################################
## 3. Coverage: mean depth and per-base depth (mosdepth v0.3.3)
######################################################################
cd <WD>
mkdir -p wgs_coverage_bt2
for b in mapping_bt2/*.mkdup.bam; do
  s=$(basename "$b" .mkdup.bam)
  mosdepth --fast-mode --threads 8 "wgs_coverage_bt2/${s}" "$b"   # no -n, so *.per-base.bed.gz is written
done

echo -e "Sample\tMeanDepth"
for f in wgs_coverage_bt2/*.mosdepth.summary.txt; do
  s=$(basename "$f" .mosdepth.summary.txt)
  awk -v s="$s" 'NR==2 {print s "\t" $4}' "$f"
done


######################################################################
## 4. Cohort SNP calling and filtering for diversity (BCFtools v1.21)
######################################################################
conda activate bcftools-env
mapfile -t BAMS < <(ls -1 <WD>/mapping_bt2/*.mkdup.bam)

bcftools mpileup --threads 16 --skip-indels -d 10000 -f <REF_FA> "${BAMS[@]}" -Ou \
| bcftools call --threads 16 --skip-variants indels --multiallelic-caller --variants-only \
    -Oz -o <WD>/snps/bcftools.snps.raw.vcf.gz
tabix -p vcf <WD>/snps/bcftools.snps.raw.vcf.gz

# Biallelic SNPs with QUAL >= 30, MQ >= 30 and <= 50% missing genotypes
# (filter chain as recorded in the output VCF header)
bcftools +fill-tags <WD>/snps/bcftools.snps.raw.vcf.gz -Ou -- -t F_MISSING,AN,AC,MAF,NS \
| bcftools view -m2 -M2 -v snps -Ou \
| bcftools filter -e 'F_MISSING>0.5 || INFO/MQ<30 || QUAL<30' -Oz -o <WD>/snps/snps_filtered_bt2.vcf.gz
tabix -p vcf <WD>/snps/snps_filtered_bt2.vcf.gz
# 18,371,386 SNPs

# Per-sample FreeBayes calls and the SNP masks for the methylation data are in
# 02_WGBS_bsseq_prep.txt, section F.


######################################################################
## 5. Genome-wide pi, Watterson's theta, Tajima's D and Ne (Table S1; VCFtools v0.1.17)
######################################################################
# Per-site statistics are divided by the full reference length (all 175 scaffolds,
# 1,651,950,171 bp); no callable-region mask is applied.
# Ne = pi / (4 mu), mu = 8.6e-9 per site per generation (Wooldridge et al. 2025).
conda activate bcftools-env
cd <WD>
V=snps/snps_filtered_bt2.vcf.gz
PREFIX=snps_filtered_bt2
MU=8.6e-9
EFF_LEN=$(awk '{s+=$2}END{print s}' <REF_FA>.fai)

vcftools --gzvcf "$V" --site-pi --out "${PREFIX}_site_pi"
vcftools --gzvcf "$V" --TajimaD ${EFF_LEN} --out "${PREFIX}_tajimaD"    # window >= scaffold: one value per scaffold
vcftools --gzvcf "$V" --freq --out "${PREFIX}_freq"

SUM_PI=$(awk 'NR>1{sum+=$3}END{printf("%.10f", sum+0)}' "${PREFIX}_site_pi.sites.pi")
PI_GENOME=$(python3 -c "print(f'{float(\"${SUM_PI}\")/${EFF_LEN}:.10f}')")
D_AVG=$(awk 'NR>1 && $4!="."{sum+=$4;cnt++}END{printf("%.6f", sum/cnt)}' "${PREFIX}_tajimaD.Tajima.D")
SITES=$(awk 'NR>1' "${PREFIX}_freq.frq" | wc -l)
N_SAMPLES=$(bcftools query -l "$V" | wc -l)
N_CHR=$((2 * N_SAMPLES))
A_N=$(python3 -c "n=${N_CHR}; print(sum(1.0/i for i in range(1,n)))")
THETA_RAW=$(python3 -c "print(f'{${SITES}/${A_N}:.10f}')")
THETA_NORM=$(python3 -c "print(f'{${THETA_RAW}/${EFF_LEN}:.12f}')")
NE=$(python3 -c "print(f'{${PI_GENOME}/(4*${MU}):.2f}')")

SUMMARY=genomewide_diversity_summary.tsv
echo -e "PREFIX\tN_SAMPLES\tSITES\tPI_genomewide\tTHETA_W_raw\tTHETA_W_norm\tTAJIMA_D_avg\tNE(mu=${MU})\tLEN_USED" > "$SUMMARY"
echo -e "${PREFIX}\t${N_SAMPLES}\t${SITES}\t${PI_GENOME}\t${THETA_RAW}\t${THETA_NORM}\t${D_AVG}\t${NE}\t${EFF_LEN}" >> "$SUMMARY"
column -t "$SUMMARY"
# N_SAMPLES 7; SITES 18371386; PI_genomewide 0.0028085918; THETA_W_norm 0.0034970319;
# TAJIMA_D_avg -0.237970; NE 81645.11; LEN_USED 1651950171
