# Command log: WGBS read processing, bisulfite-conversion check, SNP masking, and BSseq object
#
# Steps A–F were run in a terminal or as SLURM jobs on the lab server; step G is R.
# Paths are placeholders (<...>). Commands are those of the job scripts run on the server;
# tool versions are those recorded in the per-sample Trim Galore and Bismark report files.
#
# WGBS libraries (Psomagen, 150-bp paired-end) and the names used in the notebooks:
#   44-mc9a  = MC09 (M44-9-1yr)     47-mc15a = MC15 (M47-15-6yr)
#   45-mc11a = MC11 (M45-11-5yr)    48-mc18a = MC18 (M48-18-8yr)
#   46-mc14a = MC14 (M46-14-4yr)    49-mc19a = MC19 (M49-19-8yr)
#                                   50-mc24a = MC24 (M50-24-1yr)
# Libraries 44–46 and 47–50 were prepared in two batches (see Methods, Table S4).

SAMPLES=(44-mc9a 45-mc11a 46-mc14a 47-mc15a 48-mc18a 49-mc19a 50-mc24a)


###############################################################################
# A) Adapter and quality trimming (Trim Galore v0.6.10, Cutadapt v2.6)
###############################################################################
# Defaults, as recorded in the trimming reports: Phred quality cutoff 20; adapter
# auto-detected (Illumina TruSeq, AGATCGGAAGAGC); stringency 1 bp; error rate 0.1;
# pairs removed if either read is < 20 bp.
# Table S4 "raw read pairs" = "Total reads processed" in each *_1 trimming report.

conda activate trim-galore
for s in "${SAMPLES[@]}"; do
  trim_galore --paired --output_dir <TRIM_DIR> \
    <RAW_DIR>/${s}_1.fastq.gz <RAW_DIR>/${s}_2.fastq.gz
done
# Output: <TRIM_DIR>/${s}_1_val_1.fq.gz, <TRIM_DIR>/${s}_2_val_2.fq.gz (+ trimming reports)


###############################################################################
# B) Bisulfite genome preparation (Bismark v0.24.2)
###############################################################################
conda activate bismark
bismark_genome_preparation --verbose <GENOME_DIR>   # folder with mytilus_californianus_genome.fasta (GCF_021869535.1)
bismark_genome_preparation --verbose <LAMBDA_DIR>   # folder with the lambda phage genome (NEB N3013 spike-in)


###############################################################################
# C) Alignment, deduplication and methylation extraction (Bismark v0.24.2)
###############################################################################
# Bismark defaults (directional library). Bowtie 2 options recorded in every PE report:
#   -q --score-min L,0,-0.2 --ignore-quals --no-mixed --no-discordant --dovetail --maxins 500
# Table S4 "cleaned read pairs" = "Sequence pairs analysed in total" in the PE report.
# Overlapping mate calls are removed (Bismark default for paired-end data).
# Run as a SLURM array, one task per library.

for s in "${SAMPLES[@]}"; do
  bismark --multicore 3 --output_dir <MAP_DIR> --genome <GENOME_DIR> \
    -1 <TRIM_DIR>/${s}_1_val_1.fq.gz -2 <TRIM_DIR>/${s}_2_val_2.fq.gz

  deduplicate_bismark --paired <MAP_DIR>/${s}_1_val_1_bismark_bt2_pe.bam --output_dir <MAP_DIR>

  bismark_methylation_extractor --paired-end --multicore 3 --gzip --genome_folder <GENOME_DIR> \
    --buffer_size 10G --cytosine_report --comprehensive --output_dir <MAP_DIR> \
    <MAP_DIR>/${s}_1_val_1_bismark_bt2_pe.deduplicated.bam

  bam2nuc --genome_folder <GENOME_DIR> --dir <MAP_DIR> <MAP_DIR>/${s}_1_val_1_bismark_bt2_pe.deduplicated.bam

  bismark2report --output <MAP_DIR> \
    --alignment_report  <MAP_DIR>/${s}_1_val_1_bismark_bt2_PE_report.txt \
    --dedup_report      <MAP_DIR>/${s}_1_val_1_bismark_bt2_pe.deduplication_report.txt \
    --splitting_report  <MAP_DIR>/${s}_1_val_1_bismark_bt2_pe.deduplicated_splitting_report.txt \
    --mbias_report      <MAP_DIR>/${s}_1_val_1_bismark_bt2_pe.deduplicated.M-bias.txt \
    --nucleotide_report <MAP_DIR>/${s}_1_val_1_bismark_bt2_pe.deduplicated.nucleotide_stats.txt
done
# Output used below: <MAP_DIR>/${s}_1_val_1_bismark_bt2_pe.deduplicated.bismark.cov.gz
# (--cytosine_report also writes the bedGraph / .bismark.cov.gz coverage file)


###############################################################################
# D) Bisulfite-conversion check on the lambda spike-in
###############################################################################
# The same trimmed reads are aligned to the lambda genome, deduplicated and extracted.
for s in "${SAMPLES[@]}"; do        # SLURM array, one task per library
  bismark --multicore 3 --output_dir <CONV_DIR> --genome <LAMBDA_DIR> \
    -1 <TRIM_DIR>/${s}_1_val_1.fq.gz -2 <TRIM_DIR>/${s}_2_val_2.fq.gz
  deduplicate_bismark --paired <CONV_DIR>/${s}_1_val_1_bismark_bt2_pe.bam --output_dir <CONV_DIR>
  bismark_methylation_extractor --paired-end --multicore 3 --gzip --genome_folder <LAMBDA_DIR> \
    --buffer_size 10G --cytosine_report --comprehensive --output_dir <CONV_DIR> \
    <CONV_DIR>/${s}_1_val_1_bismark_bt2_pe.deduplicated.bam
done

# Conversion rate (Table S4) = 100 x (1 - methylated C / all C), summed over the CpG, CHG
# and CHH contexts of each lambda splitting report.
for r in <CONV_DIR>/*_pe.deduplicated_splitting_report.txt; do
  awk -F'\t' -v s="$(basename "${r%%_1_val_1*}")" '
    /^Total methylated C.s in (CpG|CHG|CHH) context/ {m += $2}
    /^Total C to T conversions in (CpG|CHG|CHH) context/ {u += $2}
    END {printf "%s\t%.1f%%\n", s, 100 * (1 - m / (m + u))}' "$r"
done
# Output: 44-mc9a 98.6%, 45-mc11a 98.7%, 46-mc14a 98.6%, 47-mc15a 99.7%,
#         48-mc18a 99.7%, 49-mc19a 99.7%, 50-mc24a 99.7%


###############################################################################
# E) Merge CpG strands (Bismark coverage2cytosine)
###############################################################################
# Output (per sample): <MAP_DIR>/<SAMPLE>_merged.cov.CpG_report.merged_CpG_evidence.cov
# (0-based; columns chr, start, end, % methylation, methylated count, unmethylated count)

cd <MAP_DIR>
find *deduplicated.bismark.cov.gz \
| xargs basename -s deduplicated.bismark.cov.gz \
| xargs -I{} coverage2cytosine \
  --genome_folder <GENOME_DIR> \
  --output "{}_merged.cov" \
  --merge_CpG \
  --zero_based \
  "{}deduplicated.bismark.cov.gz"

# Per-sample tables with >= 4 reads, used as the WGBS input of notebook 03 (data/wgbs/):
for f in *merged_CpG_evidence.cov; do
  STEM=$(basename "${f}" .CpG_report.merged_CpG_evidence.cov)
  awk -F $'\t' 'BEGIN {OFS = FS} {if ($5+$6 >= 4) {print $1, $2, $3, $4, $5, $6}}' "${f}" \
  > "${STEM}"_4x.tab
done
# These tables were then SNP-filtered and gzipped (*_merged.cov_4x.tab.gz); see F(4).


###############################################################################
# F) SNP masks from the WGS data
###############################################################################
# WGS BAMs (<WGS_BAM_DIR>/<SAMPLE>.mkdup.bam) come from 01_WGS_diversity.txt, steps 1–2.

# (1) Per-sample FreeBayes calls (FreeBayes v1.3.2, default settings, no quality filter;
#     SLURM array, one task per BAM)
conda activate freebayes
cd <WGS_BAM_DIR>
for BAM in *.mkdup.bam; do
  SAMPLE="${BAM%%.mkdup.bam}"
  freebayes -f <REF_FA> "$BAM" > "${SAMPLE}.raw.vcf" 2> "${SAMPLE}.freebayes.log"
  bgzip -f "${SAMPLE}.raw.vcf"
  tabix -p vcf "${SAMPLE}.raw.vcf.gz"
done

# (2) Merge into one cohort VCF (BCFtools v1.21; no further quality filtering, so masking is inclusive)
conda activate bcftools-env
bcftools merge mc*.raw.vcf.gz -Oz -o <WD>/snps/freebayes_relaxed_bt2.vcf.gz
tabix -f -p vcf <WD>/snps/freebayes_relaxed_bt2.vcf.gz

# (3) CpG-affecting mask: multiallelic records split, then C>T and G>A SNPs written as
#     1-bp BED intervals (0-based), sorted and merged (bedtools v2.31.1).
#     This is the mask applied to the BSseq object in G.
cd <WD>
VCF=snps/freebayes_relaxed_bt2.vcf.gz
TMP=snps/mask_partner.tmp.bed
OUT=snps/mask_all_snps_partner.bed
bcftools norm -m -any -Ou "$VCF" \
| bcftools view -v snps -i '((REF="C" && ALT="T") || (REF="G" && ALT="A"))' -Ou \
| bcftools query -f '%CHROM\t%POS0\t%POS\n' \
| LC_ALL=C sort -k1,1 -k2,2n > "$TMP"
bedtools merge -i "$TMP" > "$OUT"
bgzip -f "$OUT"                                  # -> snps/mask_all_snps_partner.bed.gz
tabix -f -p bed "${OUT}.gz"

# (4) SNP filter for the notebook 03 tables (E). These tables were filtered earlier (2024)
#     against a bcftools cohort SNP set, removing CpGs whose C position matched a called SNP.
#     SNP set: the trimmed WGS reads (01, step 1) aligned with Bismark v0.24.2 to <GENOME_DIR>
#     (bismark --genome <GENOME_DIR> -1 ... -2 ... --bam), then samtools sort -n, fixmate -m, sort,
#     markdup -r and index, giving <WGS_BAM_DIR_2024>/<SAMPLE>_sorted_markdup.bam:
bcftools mpileup --threads 8 --skip-indels -d 10000 -f <REF_FA> -o snps/snps-raw.bcf <WGS BAMs>
bcftools index snps/snps-raw.bcf
bcftools call --threads 8 --skip-variants indels --multiallelic-caller --variants-only \
  -O z -o snps/snps.vcf.gz snps/snps-raw.bcf
gunzip -k snps/snps.vcf.gz

R --vanilla <<'RSCRIPT'

vcf_file <- "snps/snps.vcf"
metadata_lines <- sum(grepl("^##", readLines(vcf_file)))

  mutate(loci = paste0(V1, "_", V2))
for (file in list.files("<MAP_DIR>", pattern = "_4x.tab$", full.names = TRUE)) {
  data <- read.csv(file, header = FALSE, sep = "\t")
  modified_data <- data %>%
    mutate(loci = paste0(V1, "_", V2 + 1)) %>%     # 0-based start -> 1-based VCF POS
    anti_join(ct, by = "loci") %>%
    select(-loci)
  write.table(modified_data, file = file.path("<SNPS_REMOVED_DIR>", basename(file)),
              sep = "\t", row.names = FALSE, quote = FALSE, col.names = FALSE)
}
RSCRIPT
gzip <SNPS_REMOVED_DIR>/*_4x.tab                  # -> data/wgbs/*_merged.cov_4x.tab.gz for notebook 03


###############################################################################
# G) R: build the BSseq object, apply the SNP mask, global methylation (Table S5)
###############################################################################
# Inputs:  <MAP_DIR>/*merged_CpG_evidence.cov (E); snps/mask_all_snps_partner.bed.gz (F3)
# Outputs: bsseq_raw_all_samples.RData       (7 samples, unmasked)
#          bsseq_snp_masked.RData            (6 samples, masked; data/ input of notebooks 05–07)
#          bsseq_snp_masked_all_samples.RData (7 samples, masked; Table S5)

R --vanilla <<'RSCRIPT'
suppressMessages({
  library(bsseq)
  library(data.table)
  library(GenomicRanges)
  library(GenomeInfoDb)
  library(rtracklayer)
})

cov_dir      <- "<MAP_DIR>"
snp_bed_path <- "snps/mask_all_snps_partner.bed.gz"
min_meth_cov <- 4

## ---- Load and merge the per-sample merged-CpG .cov files ----
cov_files    <- list.files(path = cov_dir, pattern = "merged_CpG_evidence\\.cov$", full.names = TRUE)
sample_names <- c("M44-9-1yr", "M45-11-5yr", "M46-14-4yr", "M47-15-6yr", "M48-18-8yr", "M49-19-8yr", "M50-24-1yr")

read_cov <- function(file, sample) {
  dt <- fread(file, header = FALSE)
  dt <- dt[, .(chr = V1, pos = V2, X = V5, N = V5 + V6)]
  setnames(dt, c("X", "N"), c(paste0("X_", sample), paste0("N_", sample)))
  return(dt)
}

cov_data <- mapply(read_cov, cov_files, sample_names, SIMPLIFY = FALSE)
combined_cov <- Reduce(function(x, y) merge(x, y, by = c("chr", "pos"), all = TRUE), cov_data)
combined_cov[is.na(combined_cov)] <- 0

## ---- Raw BSseq object (7 samples) ----
M   <- as.matrix(combined_cov[, grep("^X_", names(combined_cov)), with = FALSE])
Cov <- as.matrix(combined_cov[, grep("^N_", names(combined_cov)), with = FALSE])
colnames(M) <- colnames(Cov) <- sub("^X_", "", colnames(combined_cov)[grep("^X_", names(combined_cov))])
BSseq_obj <- BSseq(chr = combined_cov$chr, pos = combined_cov$pos, M = M, Cov = Cov, sampleNames = colnames(M))
save(BSseq_obj, file = "bsseq_raw_all_samples.RData")

## ---- Remove the WGBS coverage outlier (fewest CpGs with >= 4 reads: M45-11-5yr = MC11) ----
cov_stats <- data.frame(
  sample = sampleNames(BSseq_obj),
  covered_4x = sapply(seq_len(ncol(BSseq_obj)), function(i) sum(getCoverage(BSseq_obj, type = "Cov")[, i] >= min_meth_cov))
)
outlier_sample <- cov_stats$sample[which.min(cov_stats$covered_4x)]
BSseq_obj_clean <- BSseq_obj[, !(sampleNames(BSseq_obj) %in% outlier_sample)]

## ---- SNP mask helper: drop CpGs overlapping the partner mask ----
mask_bsseq <- function(bs, mask) {
  shared <- intersect(seqlevels(bs), seqlevels(mask))
  mask <- keepSeqlevels(mask, shared, pruning.mode = "coarse")
  bs   <- keepSeqlevels(bs,   shared, pruning.mode = "coarse")
  genome(mask) <- genome(bs)
  hits <- countOverlaps(granges(bs), mask, type = "any")
  cat("Total loci:", length(hits), "\nOverlap:", sum(hits > 0), "\nRetained:", sum(hits == 0), "\n")
  bs[hits == 0, ]
}
mask <- import(snp_bed_path)

## ---- 6-sample masked object (input of notebooks 05, 06, 07) ----
BSseq_obj_masked <- mask_bsseq(BSseq_obj_clean, mask)
save(BSseq_obj_masked, file = "bsseq_snp_masked.RData")
# Total loci: 27531884 / Overlap: 429099 / Retained: 27102785

## ---- 7-sample masked object and global methylation (Table S5) ----
BSseq_obj_masked7 <- mask_bsseq(BSseq_obj, mask)
stopifnot(ncol(BSseq_obj_masked7) == 7)
save(BSseq_obj_masked7, file = "bsseq_snp_masked_all_samples.RData")

meth_mat <- getMeth(BSseq_obj_masked7, type = "raw")   # mCG/CG per CpG and sample (NA where uncovered)
global_mean <- mean(meth_mat, na.rm = TRUE) * 100
global_per_sample <- colMeans(meth_mat, na.rm = TRUE) * 100
print(global_mean)
print(global_per_sample)
